\cvsection{RESEARCH EXPERIENCE}
\researchentry{OV-OrthKD: Open-Vocabulary Audio-Visual Event Localization}{ACM MM 2026; co-first author}
\begin{itemize}
\item Routed strong visual supervision to the localization path and weaker audio supervision to an orthogonal auxiliary path; used text prototypes to support unseen event categories.
\item Reached \textbf{0.816 segment average precision} on OV-AVEBench. F1@0.5 improved over the official fine-tuning baseline by \textbf{2.7 points} overall and \textbf{3.4 points} on unseen categories.
\end{itemize}
\researchentry{OrthKD: Lightweight Diabetic Retinopathy Screening}{IJCAI 2026; co-first author}
\begin{itemize}
\item Combined full supervision from a stronger teacher with representation-level transfer from a weaker teacher; orthogonal student projections preserved complementary clinical information.
\item Used a combined pool of \textbf{132,049} retinal images. A \textbf{5.4M}-parameter student reached \textbf{0.885 QWK} (quadratic weighted kappa) on EyePACS and improved zero-shot Messidor-2 QWK from \textbf{0.507 to 0.728}.
\end{itemize}
\researchentry{DeepYOLO: Visual Reasoning through Symbolic Scene Descriptions}{IJCNN 2026; corresponding author}
\begin{itemize}
\item Proposed the initial idea and worked with a junior student, who led implementation and experiments, to connect an object detector with a text-only language model through compact scene reports.
\item The pipeline reached \textbf{42.3\%} visual question answering accuracy on a stratified \textbf{5,000-image VQAv2 subset}, versus \textbf{38.2\%} for scene-graph serialization, using \textbf{1/2.5 as many prompt tokens as that baseline}.
\end{itemize}
\researchentry{Generated Image Evaluation}{2026}
\begin{itemize}
\item Studied whether generated images satisfy individual requirements in text prompts, using human annotations to separate different types of faithfulness errors.
\item Compared lightweight evaluators under matched conditions; examined confidence calibration, supporting image regions, and transfer across generators and datasets.
\end{itemize}
\researchentry{Reliability in Model Interpretability}{2026}
\begin{itemize}
\item Studied which functional claims observations from neural-network interventions can support, including cases where matching observed outputs leaves other behavior unresolved.
\item Combined mathematical analysis with controlled language-model tests to examine additional measurements and failures on previously unmeasured responses.
\end{itemize}
\researchentry{Understanding and Generation in Multimodal Models}{2026}
\begin{itemize}
\item Audited whether training for visual understanding changes image generation in unified multimodal models, focusing on counting and spatial relations.
\item Used paired model states and controlled comparisons to distinguish gains in answering questions, effects on generation, and transfer of the requested visual attribute.
\end{itemize}
\researchentry{Reliability of Neural Operator Predictions}{2026}
\begin{itemize}
\item Studied risk indicators available within one prediction pass of a neural operator, preserving the predicted physical field and avoiding a separate learned risk head.
\item Tested how the scoring representation affects error ranking and acceptance decisions across physical tasks; compared screening quality and computational cost with alternative indicators.
\end{itemize}
